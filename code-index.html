论文附件 1 的原始导入脚本（Excel → Neo4j）

这是论文二维码里那张代码图的内容：用 pandas 读五列表，再用 py2neo 写入本地 Neo4j。这里把口令隐去，仅作形态说明。

from py2neo import Graph, Node
import pandas as pd

def init_py2neo():
    t1 = pd.read_excel('工作簿1.xlsx').drop_duplicates()
    g = Graph(profile="http://127.0.0.1:7474",
              auth=("neo4j", "***"))
    g.delete_all()

    is_create = []
    for index, i in t1[['sub_type', 'sub_name']].iterrows():
        if i['sub_name'] not in is_create:
            is_create.append(i['sub_name'])
            node = Node(*labels: i['sub_type'], name=i['sub_name'])
            g.create(node)
    # 对象节点同理（obj_type / obj_name）
    for index, i in t1.iterrows():
        query = ("match(p),(q) where p.name='%s' and q.name='%s' "
                 "create (p)-[rel:%s]->(q)") % (
                 i['sub_name'], i['obj_name'], i['rel_name'])
        try:
            g.run(query)
        except Exception as e:
            print(e)
